// Isolated CUDA-pointer NVENC consumer. Legacy open() keeps the low-latency
// CONSTQP transport probe. open_ex() adds a single-buffer production-like HQ
// VBR/CQ policy and NV12/ARGB inputs. Never changes application encoder policy.
#define WIN32_LEAN_AND_MEAN
#include <windows.h>
#include <cstdint>
#include <cstring>
#include "nvEncodeAPI.h"

namespace {
NV_ENCODE_API_FUNCTION_LIST api{};
void* encoder = nullptr;
NV_ENC_REGISTERED_PTR resource = nullptr;
NV_ENC_OUTPUT_PTR bitstream = nullptr;
HMODULE module = nullptr;
unsigned width = 0, height = 0, stride = 0;
NV_ENC_BUFFER_FORMAT format = NV_ENC_BUFFER_FORMAT_ABGR;
int last_error = 0;
bool ok(NVENCSTATUS result) { last_error = int(result); return result == NV_ENC_SUCCESS; }

NV_ENC_BUFFER_FORMAT buffer_format(unsigned value) {
    if (value == 1) return NV_ENC_BUFFER_FORMAT_NV12;
    if (value == 2) return NV_ENC_BUFFER_FORMAT_ARGB;
    return NV_ENC_BUFFER_FORMAT_ABGR;
}

int encode_picture(unsigned index, void* bytes, unsigned capacity, unsigned* size) {
    if (!encoder || !resource || !bytes || !size) return 0;
    NV_ENC_MAP_INPUT_RESOURCE mapped{};
    mapped.version = NV_ENC_MAP_INPUT_RESOURCE_VER;
    mapped.registeredResource = resource;
    if (!ok(api.nvEncMapInputResource(encoder, &mapped))) return 0;
    NV_ENC_PIC_PARAMS pic{};
    pic.version = NV_ENC_PIC_PARAMS_VER;
    pic.outputBitstream = bitstream;
    pic.pictureStruct = NV_ENC_PIC_STRUCT_FRAME;
    pic.frameIdx = index;
    pic.inputTimeStamp = index;
    pic.inputDuration = 1;
    pic.inputBuffer = mapped.mappedResource;
    pic.bufferFmt = mapped.mappedBufferFmt;
    pic.inputWidth = width;
    pic.inputHeight = height;
    pic.inputPitch = stride;
    if (index == 0) pic.encodePicFlags = NV_ENC_PIC_FLAG_FORCEIDR | NV_ENC_PIC_FLAG_OUTPUT_SPSPPS;
    NVENCSTATUS status = api.nvEncEncodePicture(encoder, &pic);
    api.nvEncUnmapInputResource(encoder, mapped.mappedResource);
    if (status == NV_ENC_ERR_NEED_MORE_INPUT) {
        last_error = int(status);
        *size = 0;
        return 1;
    }
    if (!ok(status)) return 0;
    NV_ENC_LOCK_BITSTREAM lock{};
    lock.version = NV_ENC_LOCK_BITSTREAM_VER;
    lock.outputBitstream = bitstream;
    lock.doNotWait = 0;
    if (!ok(api.nvEncLockBitstream(encoder, &lock))) return 0;
    *size = lock.bitstreamSizeInBytes;
    bool copied = *size <= capacity;
    if (copied && *size) memcpy(bytes, lock.bitstreamBufferPtr, *size);
    if (!copied) last_error = -3;
    api.nvEncUnlockBitstream(encoder, bitstream);
    return copied ? 1 : 0;
}
}

extern "C" __declspec(dllexport) int probe_encoder_error() { return last_error; }

extern "C" __declspec(dllexport) void probe_encoder_close() {
    if (encoder) {
        if (bitstream) api.nvEncDestroyBitstreamBuffer(encoder, bitstream);
        if (resource) api.nvEncUnregisterResource(encoder, resource);
        api.nvEncDestroyEncoder(encoder);
    }
    encoder = nullptr; resource = nullptr; bitstream = nullptr;
    if (module) FreeLibrary(module);
    module = nullptr;
    width = height = stride = 0;
    format = NV_ENC_BUFFER_FORMAT_ABGR;
}

extern "C" __declspec(dllexport) int probe_encoder_open_ex(
    void* context, void* pointer, unsigned w, unsigned h, unsigned pitch,
    unsigned fmt, unsigned policy, unsigned fps_num, unsigned fps_den, unsigned cq) {
    if (encoder || !context || !pointer || !w || !h) return 0;
    format = buffer_format(fmt);
    unsigned min_pitch = format == NV_ENC_BUFFER_FORMAT_NV12 ? w : w * 4;
    if (pitch < min_pitch) return 0;
    if (!fps_num || !fps_den) { fps_num = 30; fps_den = 1; }
    if (cq > 51) cq = 19;
    module = LoadLibraryW(L"nvEncodeAPI64.dll");
    if (!module) { last_error = -1; return 0; }
    auto create = reinterpret_cast<NVENCSTATUS (NVENCAPI *)(NV_ENCODE_API_FUNCTION_LIST*)>(
        GetProcAddress(module, "NvEncodeAPICreateInstance"));
    if (!create) { last_error = -2; return 0; }
    api = {}; api.version = NV_ENCODE_API_FUNCTION_LIST_VER;
    if (!ok(create(&api))) { probe_encoder_close(); return 0; }
    NV_ENC_OPEN_ENCODE_SESSION_EX_PARAMS open{};
    open.version = NV_ENC_OPEN_ENCODE_SESSION_EX_PARAMS_VER;
    open.deviceType = NV_ENC_DEVICE_TYPE_CUDA; open.device = context; open.apiVersion = NVENCAPI_VERSION;
    if (!ok(api.nvEncOpenEncodeSessionEx(&open, &encoder))) { probe_encoder_close(); return 0; }
    NV_ENC_TUNING_INFO tuning = policy
        ? NV_ENC_TUNING_INFO_HIGH_QUALITY
        : NV_ENC_TUNING_INFO_LOW_LATENCY;
    NV_ENC_PRESET_CONFIG preset{};
    preset.version = NV_ENC_PRESET_CONFIG_VER; preset.presetCfg.version = NV_ENC_CONFIG_VER;
    if (!ok(api.nvEncGetEncodePresetConfigEx(encoder, NV_ENC_CODEC_H264_GUID, NV_ENC_PRESET_P5_GUID,
        tuning, &preset))) { probe_encoder_close(); return 0; }
    auto config = preset.presetCfg;
    // Single registered CUDA pointer cannot satisfy lookahead/B-frame delay.
    config.frameIntervalP = 1;
    config.gopLength = 120;
    config.rcParams.enableLookahead = 0;
    config.rcParams.lookaheadDepth = 0;
    config.rcParams.zeroReorderDelay = 1;
    if (policy) {
        config.rcParams.rateControlMode = NV_ENC_PARAMS_RC_VBR;
        config.rcParams.averageBitRate = 0;
        config.rcParams.maxBitRate = 0;
        config.rcParams.constQP = {cq, cq, cq};
        config.rcParams.targetQuality = static_cast<uint8_t>(cq);
        auto& vui = config.encodeCodecConfig.h264Config.h264VUIParameters;
        vui.videoSignalTypePresentFlag = 1;
        vui.videoFormat = NV_ENC_VUI_VIDEO_FORMAT_UNSPECIFIED;
        vui.videoFullRangeFlag = 0;
        vui.colourDescriptionPresentFlag = 1;
        vui.colourPrimaries = NV_ENC_VUI_COLOR_PRIMARIES_BT470BG;
        vui.transferCharacteristics = NV_ENC_VUI_TRANSFER_CHARACTERISTIC_SMPTE170M;
        vui.colourMatrix = NV_ENC_VUI_MATRIX_COEFFS_BT470BG;
    } else {
        config.rcParams.rateControlMode = NV_ENC_PARAMS_RC_CONSTQP;
        config.rcParams.constQP = {cq, cq, cq};
    }
    NV_ENC_INITIALIZE_PARAMS init{};
    init.version = NV_ENC_INITIALIZE_PARAMS_VER;
    init.encodeGUID = NV_ENC_CODEC_H264_GUID; init.presetGUID = NV_ENC_PRESET_P5_GUID;
    init.encodeWidth = init.darWidth = w; init.encodeHeight = init.darHeight = h;
    init.frameRateNum = fps_num; init.frameRateDen = fps_den;
    init.enablePTD = 1; init.enableEncodeAsync = 0; init.encodeConfig = &config;
    init.tuningInfo = tuning;
    if (!ok(api.nvEncInitializeEncoder(encoder, &init))) { probe_encoder_close(); return 0; }
    NV_ENC_REGISTER_RESOURCE reg{};
    reg.version = NV_ENC_REGISTER_RESOURCE_VER;
    reg.resourceType = NV_ENC_INPUT_RESOURCE_TYPE_CUDADEVICEPTR;
    reg.resourceToRegister = pointer; reg.width = w; reg.height = h; reg.pitch = pitch;
    reg.bufferFormat = format; reg.bufferUsage = NV_ENC_INPUT_IMAGE;
    if (!ok(api.nvEncRegisterResource(encoder, &reg))) { probe_encoder_close(); return 0; }
    resource = reg.registeredResource;
    NV_ENC_CREATE_BITSTREAM_BUFFER buffer{}; buffer.version = NV_ENC_CREATE_BITSTREAM_BUFFER_VER;
    if (!ok(api.nvEncCreateBitstreamBuffer(encoder, &buffer))) { probe_encoder_close(); return 0; }
    bitstream = buffer.bitstreamBuffer; width = w; height = h; stride = pitch;
    return 1;
}

extern "C" __declspec(dllexport) int probe_encoder_open(void* context, void* pointer, unsigned w, unsigned h, unsigned pitch) {
    return probe_encoder_open_ex(context, pointer, w, h, pitch, 0, 0, 30, 1, 19);
}

extern "C" __declspec(dllexport) int probe_encoder_frame(unsigned index, void* bytes, unsigned capacity, unsigned* size) {
    return encode_picture(index, bytes, capacity, size);
}

extern "C" __declspec(dllexport) int probe_encoder_flush(void* bytes, unsigned capacity, unsigned* size) {
    // Single-buffer sessions disable B-frames and lookahead, so each input
    // already produced a packet. EOS+LockBitstream crashed this driver.
    if (!encoder || !size) return 0;
    *size = 0;
    last_error = 0;
    (void)bytes; (void)capacity;
    return 1;
}
